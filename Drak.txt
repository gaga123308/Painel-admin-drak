-- Função universal e blindada para enviar mensagens
local function EnviarMensagemChat(texto)
    task.spawn(function()
        pcall(function()
            local channels = TextChatService:FindFirstChild("TextChannels")
            if channels then
                local generalChannel = channels:FindFirstChild("RBXGeneral")
                if generalChannel then
                    generalChannel:SendAsync(texto)
                    return
                end
                for _, child in ipairs(channels:GetChildren()) do
                    if child:IsA("TextChannel") then
                        child:SendAsync(texto)
                        return
                    end
                end
            end
            
            local defaultChatSystemChatEvents = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
            if defaultChatSystemChatEvents then
                local sayMessageRequest = defaultChatSystemChatEvents:FindFirstChild("SayMessageRequest")
                if sayMessageRequest then
                    sayMessageRequest:FireServer(texto, "All")
                    return
                end
            end
        end)
    end)
end

-- Função para criar tags RGB
local function criarTagNaCabeca(player, textoTag)
    if not player then return end
    local textLabelRef = nil
    local billboardRef = nil

    local function apply()
        local char = player.Character
        if not char then return end
        local head = char:FindFirstChild("Head")
        if not head then return end

        local old = head:FindFirstChild("DrakTagSystem")
        if old then old:Destroy() end

        local gui = Instance.new("BillboardGui")
        gui.Name = "DrakTagSystem"
        gui.Size = UDim2.new(0, 200, 0, 50)
        gui.StudsOffset = Vector3.new(0, 3, 0)
        gui.AlwaysOnTop = true
        gui.Adornee = head
        gui.Enabled = TagsGlobaisAtivas
        gui.Parent = head

        local text = Instance.new("TextLabel")
        text.Size = UDim2.new(1, 0, 1, 0)
        text.BackgroundTransparency = 1
        text.Font = Enum.Font.GothamBold
        text.TextScaled = true
        text.TextStrokeTransparency = 0.2
        text.TextStrokeColor3 = Color3.new(0, 0, 0)
        text.Text = textoTag
        text.Parent = gui
        
        textLabelRef = text
        billboardRef = gui
    end

    pcall(apply)
    player.CharacterAdded:Connect(function()
        task.wait(0.4)
        pcall(apply)
    end)

    task.spawn(function()
        while true do
            if billboardRef then
                billboardRef.Enabled = TagsGlobaisAtivas
            end
            if textLabelRef and textLabelRef.Parent and TagsGlobaisAtivas then
                local hue = tick() % 5 / 5
                textLabelRef.TextColor3 = Color3.fromHSV(hue, 1, 1)
            end
            task.wait(0.2)
        end
    end)
end

local function AtribuirTagCorreta(player)
    if not player then return end
    local nome = player.Name
    local cargo = CargosGlobais[nome]

    if cargo then
        criarTagNaCabeca(player, cargo)
    elseif AutorizadosChat[nome] then
        criarTagNaCabeca(player, "user Drak")
    end
end

local function AtualizarTodasAsTags(estado)
    TagsGlobaisAtivas = estado
    for _, player in ipairs(Players:GetPlayers()) do
        if player.Character then
            local head = player.Character:FindFirstChild("Head")
            if head then
                local gui = head:FindFirstChild("DrakTagSystem")
                if gui then
                    gui.Enabled = estado
                end
            end
        end
    end
end

local function KillPlayer(player)
    if player and player.Character and player.Character:FindFirstChild("Humanoid") then
        player.Character.Humanoid.Health = 0
    end
end

local function FirePlayer(player)
    if player and player.Character and player.Character:FindFirstChild("Head") and player.Character:FindFirstChild("Humanoid") then
        local fire = Instance.new("Fire")
        fire.Size = 15; fire.Heat = 25; fire.Parent = player.Character.Head
        player.Character.Humanoid.Health = 0
        task.delay(3, function() if fire then fire:Destroy() end end)
    end
end

local function SitPlayer(player)
    if player and player.Character and player.Character:FindFirstChild("Humanoid") then
        player.Character.Humanoid.Sit = true
    end
end

local function FloatPlayer(player)
    if player and player.Character and player.Character:FindFirstChild("HumanoidRootPart") and player.Character:FindFirstChild("Humanoid") then
        local hrp = player.Character.HumanoidRootPart
        hrp.Anchored = true
        local tween = TweenService:Create(hrp, TweenInfo.new(3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {CFrame = hrp.CFrame + Vector3.new(0, 40, 0)})
        tween:Play()
        tween.Completed:Connect(function()
            hrp.Anchored = false
            player.Character.Humanoid.Health = 0
        end)
    end
end

local function RagdollPlayer(player)
    if player and player.Character then
        local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
        if humanoid then humanoid.PlatformStand = true; RagdollPlayers[player] = true end
    end
end

local function UnragdollPlayer(player)
    if player and player.Character then
        local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
        if humanoid then humanoid.PlatformStand = false; RagdollPlayers[player] = nil end
    end
end

local function CreateJail(player)
    if not player or Jajail[player] then return end
    local character = player.Character
    if character and character:FindFirstChild("HumanoidRootPart") then
        local HRP = character.HumanoidRootPart
        local cageModel = Instance.new("Model")
        cageModel.Name = "JaulaDeAdmin_" .. player.Name
        cageModel.Parent = workspace

        local function createWall(size, posOffset)
            local wall = Instance.new("Part")
            wall.Size = size; wall.Anchored = true; wall.CanCollide = true
            wall.Transparency = 0.3; wall.Color = Color3.fromRGB(255, 204, 0)
            wall.Material = Enum.Material.ForceField
            wall.CFrame = HRP.CFrame * posOffset
            wall.Parent = cageModel
        end

        createWall(Vector3.new(10, 0.5, 10), CFrame.new(0, -5, 0))
        createWall(Vector3.new(10, 0.5, 10), CFrame.new(0, 5, 0))
        createWall(Vector3.new(10, 10, 0.5), CFrame.new(0, 0, -5))
        createWall(Vector3.new(10, 10, 0.5), CFrame.new(0, 0, 5))
        createWall(Vector3.new(0.5, 10, 10), CFrame.new(5, 0, 0))
        createWall(Vector3.new(0.5, 10, 10), CFrame.new(-5, 0, 0))

        Jajail[player] = cageModel
    end
end

local function RemoveJail(player)
    if player and Jajail[player] then
        Jajail[player]:Destroy()
        Jajail[player] = nil
    end
end

local function FreezePlayer(player)
    if player and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
        player.Character.HumanoidRootPart.Anchored = true
        FrozenPlayers[player] = true
    end
end

local function UnfreezePlayer(player)
    if player and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
        player.Character.HumanoidRootPart.Anchored = false
        FrozenPlayers[player] = nil
    end
end

local function BringPlayer(targetPlayer)
    if targetPlayer and targetPlayer.Character and LocalPlayer.Character then
        local localHrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        local targetHrp = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
        if localHrp and targetHrp then
            targetHrp.CFrame = localHrp.CFrame + Vector3.new(3, 0, 0)
        end
    end
end

local SelectedPlayerName = ""

local HasTriggeredVerifique = false

local function getPlayerNames()
    local names = {}
    for _, player in ipairs(Players:GetPlayers()) do

local function ProcessarMensagemChat(msgText, authorName)
    if not msgText or not authorName then return end
    local msgLower = msgText:lower()

    -- 1. Qualquer um pode responder ao ;verifique
    if msgLower:match("^;verifique") then
        if authorName ~= LocalPlayer.Name and not HasTriggeredVerifique then
            HasTriggeredVerifique = true
            task.delay(math.random(0.1, 0.3), function()
                EnviarMensagemChat("Drak_####")
            end)
            task.delay(4, function() HasTriggeredVerifique = false end)
        end
    end

    -- Sistema de tags liberado para sincronização
    if msgLower:match("^;tag all") then AtualizarTodasAsTags(true) end
    if msgLower:match("^;untag all") then AtualizarTodasAsTags(false) end

    if msgText:match("[Dd]rak_") then
        local targetPlayer = Players:FindFirstChild(authorName)
        if targetPlayer then
            if not CargosGlobais[authorName] then AutorizadosChat[authorName] = true end
            AtribuirTagCorreta(targetPlayer)
        end
    end

    -- 2. TRAVA DE SEGURANÇA: Da daqui para baixo, APENAS ADMINS podem disparar comandos pelo chat!
    if not IsUserAdmin(authorName) then
        return
    end

    for index, data in ipairs(JUMPSCARES) do
        if msgLower:match("^" .. data.Name) then
            local targetName = msgText:match("^" .. data.Name .. "%s+(.+)")
            if targetName then
                targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
                if targetName:lower() == LocalPlayer.Name:lower() then
                    TriggerJumpscare(index)
                end
            end
        end
    end

    if msgLower:match("^;kill") and not msgLower:match("^;killplus") then
        local targetName = msgText:match("^;kill%s+(.+)")
        if targetName then
            targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
            local targetPlayer = Players:FindFirstChild(targetName)
            if targetPlayer then KillPlayer(targetPlayer) end
        end
    end

    for _, playerName in ipairs(getPlayerNames()) do
        if msgText:match(";kick%s+" .. playerName) then
            if LocalPlayer.Name:lower() == playerName:lower() then
                LocalPlayer:Kick("equipe Drak")
            end
        end
    end

    if msgLower:match("^;fire") then
        local targetName = msgText:match("^;fire%s+(.+)")
        if targetName then
            targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
            local targetPlayer = Players:FindFirstChild(targetName)
            if targetPlayer then FirePlayer(targetPlayer) end
        end
    end

    if msgLower:match("^;sit") then
        local targetName = msgText:match("^;sit%s+(.+)")
        if targetName then
            targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
            local targetPlayer = Players:FindFirstChild(targetName)
            if targetPlayer then SitPlayer(targetPlayer) end
        end
    end

    if msgLower:match("^;float") then
        local targetName = msgText:match("^;float%s+(.+)")
        if targetName then
            targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
            local targetPlayer = Players:FindFirstChild(targetName)
            if targetPlayer then FloatPlayer(targetPlayer) end
        end
    end

    if msgLower:match("^;ragdoll") and not msgLower:match("^;unragdoll") then
        local targetName = msgText:match("^;ragdoll%s+(.+)")
        if targetName then
            targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
            local targetPlayer = Players:FindFirstChild(targetName)
            if targetPlayer then RagdollPlayer(targetPlayer) end
        end
    end

    if msgLower:match("^;unragdoll") then
        local targetName = msgText:match("^;unragdoll%s+(.+)")
        if targetName then
            targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
            local targetPlayer = Players:FindFirstChild(targetName)
            if targetPlayer then UnragdollPlayer(targetPlayer) end
        end
    end

    -- CORREÇÃO DO ;CHT (Faz qualquer painel aberto - 1º ou 2º plano - forçar o alvo a falar)
    if msgLower:match("^;cht") then
        local targetUser, saidaMsg = msgText:match("^;cht%s+([^%s]+)%s+(.+)")
        if targetUser and saidaMsg then
            targetUser = targetUser:gsub("^%s*(.-)%s*$", "%1")
            if targetUser:lower() == LocalPlayer.Name:lower() then
                EnviarMensagemChat(saidaMsg)
            end
        end
    end

    if msgLower:match("^;jail") then
        local targetName = msgText:match("^;jail%s+(.+)")
        if targetName then
            targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
            local targetPlayer = Players:FindFirstChild(targetName)
            if targetPlayer then CreateJail(targetPlayer) end
        end
    end

    if msgLower:match("^;unjail") then
        local targetName = msgText:match("^;unjail%s+(.+)")
        if targetName then
            targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
            local targetPlayer = Players:FindFirstChild(targetName)
            if targetPlayer then RemoveJail(targetPlayer) end
        end
    end

    if msgLower:match("^;freeze") then
        local targetName = msgText:match("^;freeze%s+(.+)")
        if targetName then
            targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
            local targetPlayer = Players:FindFirstChild(targetName)
            if targetPlayer then FreezePlayer(targetPlayer) end
        end
    end

    if msgLower:match("^;unfreeze") then
        local targetName = msgText:match("^;unfreeze%s+(.+)")
        if targetName then
            targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
            local targetPlayer = Players:FindFirstChild(targetName)
            if targetPlayer then UnfreezePlayer(targetPlayer) end
        end
    end

    if msgLower:match("^;bring") then
        local targetName = msgText:match("^;bring%s+(.+)")
        if targetName then
            targetName = targetName:gsub("^%s*(.-)%s*$", "%1")
            local targetPlayer = Players:FindFirstChild(targetName)
            if targetPlayer then BringPlayer(targetPlayer) end
        end
    end

    if msgLower:match("^;goner") then
        TransformIntoGoner()
    end
end

-- Conectar chat moderno
local function ConectarCanal(canal)
    if not canal or not canal.IsA or not canal:IsA("TextChannel") then return end
    canal.MessageReceived:Connect(function(msg)
        local text = msg.Text
        local source = msg.TextSource and msg.TextSource.Name
        if text and source then ProcessarMensagemChat(text, source) end
    end)
end

pcall(function()
    for _, ch in pairs(TextChatService.TextChannels:GetChildren()) do ConectarCanal(ch) end
    TextChatService.TextChannels.ChildAdded:Connect(function(ch) ConectarCanal(ch) end)
end)

-- Conectar chat legado (essencial em segundo plano)
pcall(function()
    local events = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
    if events then
        local onMessageDoneFiltering = events:FindFirstChild("OnMessageDoneFiltering")
        if onMessageDoneFiltering then
            onMessageDoneFiltering.OnClientEvent:Connect(function(messageData)
                if messageData and messageData.FromSpeaker and messageData.Message then
                    local speaker = Players:FindFirstChild(messageData.FromSpeaker)
                    if speaker then
                        ProcessarMensagemChat(messageData.Message, speaker.Name)
                    end
                end
            end)
        end
    end
end)
