local function CarregarUsuariosGitHub()
    local success, response = pcall(function()
        return game:HttpGet(GITHUB_WHITELIST_URL)
    end)

    if success and response then
        local func, err = loadstring(response)
        if func then
            local result = func()
            if type(result) == "table" then
                UsuariosGitHub = result
                print("[Drak Hub] Whitelist carregada com sucesso!")
            else
                warn("[Drak Hub] O retorno do script do GitHub não é uma tabela!")
            end
        else
            warn("[Drak Hub] Erro ao compilar o script do GitHub: " .. tostring(err))
        end
    else
        warn("[Drak Hub] Erro ao baixar a whitelist do GitHub!")
    end
end

--// Sistema de Tags baseado no Cargo
local function ObterTextoTag(player)
    local cargo = UsuariosGitHub[player.Name]

    if cargo == "Dono" then
        return "Dono Drak"
    elseif cargo == "Sub dono" then
        return "Sub dono Drak"
    elseif cargo == "Influenciador" then
        return "YT Influenciador"
    elseif cargo == "Admin" then
        return "usuário admin"
    elseif cargo == "Moderador" then
        return "Moderador Drak"
    elseif cargo == "Staff" then
        return "Staff Drak"
    else
        return nil
    end
end

local function createSpecialTag(player, extraText)
    if not player then return end

    local function apply()
        local char = player.Character
        if not char then return end
        local head = char:FindFirstChild("Head")
        if not head then return end

        local cargoTag = ObterTextoTag(player)
        if not cargoTag and not extraText then return end

        local old = head:FindFirstChild("DrakTagContainer") or head:FindFirstChild("DrakTag") or head:FindFirstChild("TagsDRAK") or head:FindFirstChild("NytheruneTag")
        if old then old:Destroy() end

        local gui = Instance.new("BillboardGui")
        gui.Name = "DrakTagContainer"
        gui.Size = UDim2.new(0, 240, 0, 80)
        gui.StudsOffset = Vector3.new(0, 3.5, 0)
        gui.AlwaysOnTop = true
        gui.Adornee = head
        gui.Parent = head

        local layout = Instance.new("UIListLayout")
        layout.SortOrder = Enum.SortOrder.LayoutOrder
        layout.HorizontalAlignment = Enum.HorizontalAlignment.Center
        layout.VerticalAlignment = Enum.VerticalAlignment.Center
        layout.Padding = UDim.new(0, 2)
        layout.Parent = gui

        -- 1. Tag Principal do Cargo (Fica em Cima)
        if cargoTag then
            local textCargo = Instance.new("TextLabel")
            textCargo.Name = "CargoTag"
            textCargo.Size = UDim2.new(1, 0, 0, 30)
            textCargo.BackgroundTransparency = 1
            textCargo.Font = Enum.Font.GothamBold
            textCargo.TextScaled = true
            textCargo.TextStrokeTransparency = 0.2
            textCargo.TextStrokeColor3 = Color3.new(0, 0, 0)
            textCargo.TextColor3 = Color3.fromRGB(255, 215, 0)
            textCargo.Text = cargoTag
            textCargo.LayoutOrder = 1
            textCargo.Parent = gui
        end

        -- 2. Tag Secundária de Usuário (Fica Embaixo)
        if extraText and (not cargoTag or extraText ~= cargoTag) then
            local textExtra = Instance.new("TextLabel")
            textExtra.Name = "ExtraTag"
            textExtra.Size = UDim2.new(1, 0, 0, 25)
            textExtra.BackgroundTransparency = 1
            textExtra.Font = Enum.Font.GothamBold
            textExtra.TextScaled = true
            textExtra.TextStrokeTransparency = 0.2
            textExtra.TextStrokeColor3 = Color3.new(0, 0, 0)
            textExtra.TextColor3 = Color3.fromRGB(0, 180, 255)
            textExtra.Text = extraText
            textExtra.LayoutOrder = 2
            textExtra.Parent = gui
        end
    end

    pcall(apply)
    player.CharacterAdded:Connect(function()
        task.wait(0.4)
        pcall(apply)
    end)

--// Função Utilitária para Verificar se o Nome Corresponde
local function NomeCorresponde(alvoDigitado, nomeJogador)
    if not alvoDigitado or alvoDigitado == "" or not nomeJogador then return false end
    alvoDigitado = alvoDigitado:lower()
    nomeJogador = nomeJogador:lower()
    return nomeJogador:sub(1, #alvoDigitado) == alvoDigitado
end

--// Função de Jaula Avançada
local function CreateAndManageJail(player)
    if not player or not player.Character then return false end
    local e = Workspace:FindFirstChild("Jail_" .. player.Name)
    if e then e:Destroy() end
    if JAILED_PLAYERS[player.Name] and JAILED_PLAYERS[player.Name].loop then
        task.cancel(JAILED_PLAYERS[player.Name].loop)
    end

    local hrp = player.Character:FindFirstChild("HumanoidRootPart")
    local hum = player.Character:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return false end

    local m = Instance.new("Model", Workspace)
    m.Name = "Jail_" .. player.Name
    local s = 8
    local p = hrp.Position
    local t = {
        Base = Instance.new("Part", m),
        Roof = Instance.new("Part", m),
        Wall1 = Instance.new("Part", m),
        Wall2 = Instance.new("Part", m),
        Wall3 = Instance.new("Part", m),
        Wall4 = Instance.new("Part", m)
    }

    t.Base.Size = Vector3.new(s, 0.5, s)
    t.Base.Position = p - Vector3.new(0, hum.HipHeight + 0.25, 0)
    t.Roof.Size = Vector3.new(s, 0.5, s)
    t.Roof.Position = t.Base.Position + Vector3.new(0, s, 0)
    t.Wall1.Size = Vector3.new(0.5, s, s)
    t.Wall1.Position = t.Base.Position + Vector3.new(s / 2, s / 2, 0)
    t.Wall2.Size = Vector3.new(0.5, s, s)
    t.Wall2.Position = t.Base.Position - Vector3.new(s / 2, -s / 2, 0)
    t.Wall3.Size = Vector3.new(s, s, 0.5)
    t.Wall3.Position = t.Base.Position + Vector3.new(0, s / 2, s / 2)
    t.Wall4.Size = Vector3.new(s, s, 0.5)
    t.Wall4.Position = t.Base.Position - Vector3.new(0, -s / 2, s / 2)

    for _, v in pairs(t) do
        v.Anchored = true
        v.CanCollide = true
        v.Material = Enum.Material.Neon
        v.Color = Color3.fromRGB(0, 200, 255)
        v.Transparency = 0.6
    end

    if player == LocalPlayer then
        local l = task.spawn(function()
            local c = t.Base.Position + Vector3.new(0, hum.HipHeight, 0)
            while JAILED_PLAYERS[player.Name] and player.Parent and player.Character and player.Character.Parent do
                local h = player.Character:FindFirstChild("HumanoidRootPart")
                if h and (h.Position - c).Magnitude > (s / 2) - 1 then
                    h.CFrame = CFrame.new(c)
                end
                task.wait(0.1)
            end
        end)
        JAILED_PLAYERS[player.Name] = { loop = l }
    else
        JAILED_PLAYERS[player.Name] = {}
    end
    return true
end

local function RemoveJail(player)
    if not player then return end
    local e = Workspace:FindFirstChild("Jail_" .. player.Name)
    if e then e:Destroy() end

    if JAILED_PLAYERS[player.Name] then
        if JAILED_PLAYERS[player.Name].loop then
            task.cancel(JAILED_PLAYERS[player.Name].loop)
        end
        JAILED_PLAYERS[player.Name] = nil
    end
end

local playerOriginalSpeed = {}

--// Processamento de Comandos (Backend / Chat / Board)
local function ProcessarComando(rawText)
    if not rawText or rawText == "" then return end

    local comando = rawText:match("^;%w+") and rawText:lower() or rawText

    -- COMANDO DE VERIFICAÇÃO VIA CESTO DE ROUPA (LaundryBasket)
    if comando == ";verifique" then
        pcall(function()
            local toolRemote = ReplicatedStorage:FindFirstChild("RE") and ReplicatedStorage.RE:FindFirstChild("1Too1l")
            if toolRemote then
                toolRemote:InvokeServer("PickingTools", "LaundryBasket")
            end
        end)
        task.delay(0.5, function()
            for _, player in ipairs(Players:GetPlayers()) do
                local function possuiLaundry()
                    return (player.Backpack and player.Backpack:FindFirstChild("LaundryBasket"))
                        or (player.Character and player.Character:FindFirstChild("LaundryBasket"))
                end
                if possuiLaundry() and player.Character then
                    createSpecialTag(player, "Usuário Drak")
                end
            end
        end)
    end
end

local function ProcessarMensagem(msgText, authorName)
    if not msgText or not authorName then return end

    local comandoLower = msgText:lower()

    -- RESPOSTA DE VERIFICAÇÃO COM CÓDIGO/CHAVE DRAK_####
    if comandoLower:match("^;verifique") then
        local randomCode = math.random(1000, 9999)
        EnviarMensagemChat("Drak_" .. tostring(randomCode))
    end

    -- DETECÇÃO DA MENSAGEM DRAK_####
    -- Adiciona a tag extra de "Usuário Drak" sem remover a tag principal do jogador se ele for Dono/Admin
    if msgText:match("[Dd]rak_%d%d%d%d") then
        local authorPlayer = Players:FindFirstChild(authorName)
        if authorPlayer then
            createSpecialTag(authorPlayer, "Usuário Drak")
        end
    end

    ProcessarComando(msgText)

    -- Apenas administradores do GitHub podem disparar os comandos abaixo
    if not UsuariosGitHub[authorName] then return end

    local authorPlayer = Players:FindFirstChild(authorName)
    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local root = character and character:FindFirstChild("HumanoidRootPart")

    local cmd, targetInput, rest = msgText:match("^;(%w+)%s*([^%s]*)%s*(.*)")
    if not cmd then return end

    cmd = cmd:lower()
    targetInput = targetInput or ""
    local souOAlvo = NomeCorresponde(targetInput, LocalPlayer.Name)

    if cmd == "kick" and souOAlvo then LocalPlayer:Kick("You got kicked by Drak Hub") end
    if cmd == "kill" and souOAlvo and character then character:BreakJoints() end
    if cmd == "killplus" and souOAlvo and character then
        character:BreakJoints()
        if root then
            for i=1,10 do
                local part = Instance.new("Part")
                part.Size = Vector3.new(10,10,10)
                part.Anchored = false
                part.CanCollide = false
                part.Material = Enum.Material.Neon
                part.BrickColor = BrickColor.Random()
                part.CFrame = root.CFrame
                part.Parent = Workspace
                local bv = Instance.new("BodyVelocity")
                bv.Velocity = Vector3.new(math.random(-50,50), math.random(20,80), math.random(-50,50))
                bv.MaxForce = Vector3.new(1e5,1e5,1e5)
                bv.Parent = part
                game.Debris:AddItem(part,3)
            end
        end
    end
    if cmd == "fling" and souOAlvo and character and root then
        local tween = TweenService:Create(root, TweenInfo.new(1, Enum.EasingStyle.Linear), {CFrame = CFrame.new(0,100000,0)})
        tween:Play()
    end
    if cmd == "freeze" and souOAlvo and humanoid then
        playerOriginalSpeed[LocalPlayer.Name:lower()] = humanoid.WalkSpeed
        humanoid.WalkSpeed = 0
    end
    if cmd == "unfreeze" and souOAlvo and humanoid then
        humanoid.WalkSpeed = playerOriginalSpeed[LocalPlayer.Name:lower()] or 16
    end
    if cmd == "bring" and souOAlvo and authorPlayer and authorPlayer.Character and authorPlayer.Character:FindFirstChild("HumanoidRootPart") and root then
        root.CFrame = authorPlayer.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
    end
    if cmd == "tp" and authorName:lower() == LocalPlayer.Name:lower() then
        for _, p in ipairs(Players:GetPlayers()) do
            if NomeCorresponde(targetInput, p.Name) then
                if p.Character and p.Character:FindFirstChild("HumanoidRootPart") and root then
                    root.CFrame = p.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
                end
                break
            end
        end
    end
    if cmd == "float" and souOAlvo and character and root and humanoid then
        task.spawn(function()
            local bv = Instance.new("BodyVelocity")
            bv.Velocity = Vector3.new(0, 30, 0)
            bv.MaxForce = Vector3.new(0, 1e5, 0)
            bv.Parent = root
            task.wait(3)
            bv:Destroy()
            character:BreakJoints()
        end)
    end
    if cmd == "jail" and souOAlvo then CreateAndManageJail(LocalPlayer) end
    if cmd == "unjail" and souOAlvo then RemoveJail(LocalPlayer) end
    if cmd == "backrooms" and souOAlvo then
        local mapID = 10581711055
        pcall(function()
            local mapa = game:GetObjects("rbxassetid://"..mapID)[1]
            mapa.Parent = Workspace
            if not mapa.PrimaryPart then
                local part = mapa:FindFirstChildWhichIsA("BasePart")
                if part then mapa.PrimaryPart = part end
            end
            mapa:SetPrimaryPartCFrame(CFrame.new(Vector3.new(4000, 7, 0)))
            if character and character:FindFirstChild("HumanoidRootPart") then
                character:MoveTo(Vector3.new(4029, 9, 24))
            end
        end)
    end
    if cmd == "jumpscare" and souOAlvo then Jumpscare() end
    for _, jData in ipairs(JUMPSCARES) do
        if cmd == jData.Name and souOAlvo then CustomJumpscare(jData.ImageId, jData.AudioId) end
    end
    if cmd == "animejumps" and souOAlvo then StartAnimeJumpscare() end
    if cmd == "unjumpsanime" and souOAlvo then StopAnimeJumpscare() end
    if cmd == "goner" and souOAlvo then TransformIntoGoner() end
    if cmd == "coolkddavatar" and souOAlvo then CoolkidAvatar() end

end

--// Conecta canais de chat
local function ConectarCanal(canal)
    if not canal or not canal.IsA or not canal:IsA("TextChannel") then return end
    canal.MessageReceived:Connect(function(msg)
        local text = msg.Text
        local source = msg.TextSource and msg.TextSource.Name
        if text and source then ProcessarMensagem(text, source) end
    end)
end

for _, ch in pairs(TextChatService.TextChannels:GetChildren()) do ConectarCanal(ch) end
TextChatService.TextChannels.ChildAdded:Connect(function(ch) ConectarCanal(ch) end)
